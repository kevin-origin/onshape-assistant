== INTERFERENCE DETECTION — BROAD SEARCH ==

Trigger interference detection in the UI, keep the panel open,
then paste this (searches entire page for anything interference-related):

JSON.stringify(Array.from(document.querySelectorAll('*')).filter(function(el){if(el.offsetWidth<5||el.offsetHeight<5)return false;var text=(el.textContent||'').toLowerCase();var title=(el.getAttribute('title')||'').toLowerCase();var cls=(el.className||'').toString().toLowerCase();return(text.includes('interfer')||title.includes('interfer')||cls.includes('interfer'))&&el.children.length<10}).map(function(el){var r=el.getBoundingClientRect();return{tag:el.tagName,cls:(el.className||'').toString().slice(0,200),text:el.textContent.trim().slice(0,200),title:el.getAttribute('title')||'',x:Math.round(r.left),y:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height),html:el.outerHTML.slice(0,400)}}).slice(0,30),null,2)

== INSERT SHEET IN DRAWING TAB ==

Open a drawing tab in Onshape, then run this in the console.
Action order: Ctrl+S (save), then click "Insert Sheet".

(function(){document.dispatchEvent(new KeyboardEvent('keydown',{key:'s',code:'KeyS',ctrlKey:true,bubbles:true}));console.log('[Observer] Ctrl+S dispatched, waiting 2s for save...');setTimeout(function(){var btn=Array.from(document.querySelectorAll('button,div[role="button"],span,div')).find(function(el){return el.textContent.trim()==='Insert sheet'||el.textContent.trim()==='Insert Sheet'||el.getAttribute('aria-label')==='Insert sheet'});if(btn){btn.click();console.log('[Observer] Clicked Insert Sheet')}else{console.log('[Observer] Insert Sheet button not found — searching broader...');var all=Array.from(document.querySelectorAll('*')).filter(function(el){return(el.textContent||'').toLowerCase().includes('insert sheet')&&el.children.length<5});console.log('[Observer] Candidates:',all.map(function(el){return{tag:el.tagName,text:el.textContent.trim().slice(0,80),cls:(el.className||'').toString().slice(0,100)}}))}},2000)})()

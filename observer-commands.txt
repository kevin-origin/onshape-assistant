== INTERFERENCE DETECTION — BROAD SEARCH ==

Trigger interference detection in the UI, keep the panel open,
then paste this (searches entire page for anything interference-related):

JSON.stringify(Array.from(document.querySelectorAll('*')).filter(function(el){if(el.offsetWidth<5||el.offsetHeight<5)return false;var text=(el.textContent||'').toLowerCase();var title=(el.getAttribute('title')||'').toLowerCase();var cls=(el.className||'').toString().toLowerCase();return(text.includes('interfer')||title.includes('interfer')||cls.includes('interfer'))&&el.children.length<10}).map(function(el){var r=el.getBoundingClientRect();return{tag:el.tagName,cls:(el.className||'').toString().slice(0,200),text:el.textContent.trim().slice(0,200),title:el.getAttribute('title')||'',x:Math.round(r.left),y:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height),html:el.outerHTML.slice(0,400)}}).slice(0,30),null,2)

== DRAWING UI — CLICK & HOVER OBSERVER ==

Open a drawing tab in Onshape, paste this in devtools console,
then hover and click around normally. It logs every element you
interact with + any tooltips/bubbles that appear. Type stop() to end.

== OPTION A: Run from parent page (injects into iframe) ==

var _iframe=document.querySelector('iframe.foreign-data-container');if(!_iframe){console.log('[Observer] No drawing iframe found')}else{var _doc=_iframe.contentDocument||_iframe.contentWindow.document;var _obsLog=[];var _obsMut=new _iframe.contentWindow.MutationObserver(function(muts){muts.forEach(function(m){m.addedNodes.forEach(function(n){if(n.nodeType!==1)return;var text=(n.textContent||'').trim().slice(0,150);var cls=(n.className||'').toString();if(cls.match(/tooltip|bubble|popover|flyout|hint/i)||n.getAttribute('role')==='tooltip'){var entry={event:'tooltip',tag:n.tagName,text:text,cls:cls.slice(0,150),html:n.outerHTML.slice(0,300)};_obsLog.push(entry);console.log('[Observer] tooltip:',entry)}})})});_obsMut.observe(_doc.body,{childList:true,subtree:true});function _obsHandler(e){var el=e.target;var r=el.getBoundingClientRect();var entry={event:e.type,tag:el.tagName,text:(el.textContent||'').trim().slice(0,80),ariaLabel:el.getAttribute('aria-label')||'',title:el.getAttribute('title')||'',cls:(el.className||'').toString().slice(0,150),x:Math.round(r.left),y:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height)};_obsLog.push(entry);console.log('[Observer] '+e.type+':',entry)}_doc.addEventListener('click',_obsHandler,true);_doc.addEventListener('mouseover',_obsHandler,true);window.stop=function(){_doc.removeEventListener('click',_obsHandler,true);_doc.removeEventListener('mouseover',_obsHandler,true);_obsMut.disconnect();console.log('[Observer] Stopped. Full log ('+_obsLog.length+' events):');console.log(JSON.stringify(_obsLog,null,2))};console.log('[Observer] Running inside iframe. Hover and click toolbar, then type stop() when done.')}

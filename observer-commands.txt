STEP 1 — Start observer, then manually create/add different tab types
(Part Studio, Assembly, Drawing, Feature Studio, Variable Studio):

window.__dc=[]; window.__obs=new MutationObserver(function(muts){muts.forEach(function(m){m.addedNodes.forEach(function(node){if(node.nodeType!==1)return;var r=node.getBoundingClientRect();window.__dc.push({tag:node.tagName,cls:(node.className||"").toString().slice(0,150),text:node.textContent.trim().slice(0,120),w:Math.round(r.width),h:Math.round(r.height),x:Math.round(r.left),y:Math.round(r.top),html:node.outerHTML.slice(0,300)})})})});window.__obs.observe(document.body,{childList:true,subtree:true});console.log("Observer started");


STEP 2 — After creating the tabs, paste this to dump all tab elements with full class info:

window.__obs.disconnect();var tabs=document.querySelectorAll('.os-tab-bar-tab');var info=[];for(var t of tabs){var n=t.querySelector('.os-tab-name');var icon=t.querySelector('svg use, [class*="icon"], [class*="svg"]');info.push({name:n?n.textContent.trim():'?',cls:t.className.toString(),isFolder:t.classList.contains('os-tab-bar-tab-group'),svgHref:(icon&&(icon.getAttribute('href')||icon.getAttribute('xlink:href')))||'none',iconCls:(icon&&icon.className.toString())||'none',allAttrs:Array.from(t.attributes).map(function(a){return a.name+'='+a.value.slice(0,80)})})};console.log(JSON.stringify(info,null,2));

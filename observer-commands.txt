== INTERFERENCE DETECTION — BROAD SEARCH ==

Trigger interference detection in the UI, keep the panel open,
then paste this (searches entire page for anything interference-related):

JSON.stringify(Array.from(document.querySelectorAll('*')).filter(function(el){if(el.offsetWidth<5||el.offsetHeight<5)return false;var text=(el.textContent||'').toLowerCase();var title=(el.getAttribute('title')||'').toLowerCase();var cls=(el.className||'').toString().toLowerCase();return(text.includes('interfer')||title.includes('interfer')||cls.includes('interfer'))&&el.children.length<10}).map(function(el){var r=el.getBoundingClientRect();return{tag:el.tagName,cls:(el.className||'').toString().slice(0,200),text:el.textContent.trim().slice(0,200),title:el.getAttribute('title')||'',x:Math.round(r.left),y:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height),html:el.outerHTML.slice(0,400)}}).slice(0,30),null,2)

== DRAWING TOOLBAR — HOVER OBSERVER ==

Open a drawing tab in Onshape, then paste this in the devtools console.
It finds toolbar buttons, hovers over each one, waits for the tooltip
bubble to appear, and logs what it finds (tag, text, aria-label, tooltip).

(function(){var buttons=Array.from(document.querySelectorAll('button,[role="button"],[data-tooltip],[aria-label]')).filter(function(el){var r=el.getBoundingClientRect();return r.width>10&&r.width<80&&r.height>10&&r.height<80&&r.top<120});console.log('[Observer] Found '+buttons.length+' toolbar-area elements');var results=[];var i=0;function next(){if(i>=buttons.length){console.log('[Observer] Done. Results:');console.table(results);console.log(JSON.stringify(results,null,2));return}var btn=buttons[i];var r=btn.getBoundingClientRect();btn.dispatchEvent(new MouseEvent('mouseenter',{bubbles:true,clientX:r.left+r.width/2,clientY:r.top+r.height/2}));btn.dispatchEvent(new MouseEvent('mouseover',{bubbles:true,clientX:r.left+r.width/2,clientY:r.top+r.height/2}));setTimeout(function(){var tooltip=document.querySelector('[class*="tooltip"],[class*="Tooltip"],[role="tooltip"],[class*="bubble"],[class*="Bubble"],[class*="popover"],[class*="Popover"]');var tipText=tooltip?tooltip.textContent.trim():'';results.push({index:i,tag:btn.tagName,text:btn.textContent.trim().slice(0,60),ariaLabel:btn.getAttribute('aria-label')||'',title:btn.getAttribute('title')||'',dataTooltip:btn.getAttribute('data-tooltip')||'',tooltip:tipText.slice(0,100),cls:(btn.className||'').toString().slice(0,120),x:Math.round(r.left),y:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height)});btn.dispatchEvent(new MouseEvent('mouseleave',{bubbles:true}));btn.dispatchEvent(new MouseEvent('mouseout',{bubbles:true}));i++;setTimeout(next,300)},800)}next()})()
